-- DOXANTU : pseudo admin affiché + correction de réservation des pseudos. -- À exécuter dans Supabase > SQL Editor après les migrations initiales. -- Ne change pas l'e-mail ni le mot de passe de connexion de l'administrateur. CREATE OR REPLACE FUNCTION public.ensure_customer_username_reservation() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = pg_catalog, public AS $$ BEGIN IF auth.uid() IS NOT NULL AND NEW.user_id <> auth.uid() THEN RAISE EXCEPTION 'Tu ne peux modifier que ton propre profil.'; END IF; NEW.username := lower(trim(NEW.username)); IF NEW.username IS NULL OR NEW.username !~ '^[a-z0-9._-]{3,24}$' THEN RAISE EXCEPTION 'Le pseudo doit contenir 3 à 24 caractères : lettres sans accent, chiffres, point, tiret ou underscore.'; END IF; INSERT INTO public.customer_usernames (username, auth_user_id) VALUES (NEW.username, NEW.user_id) ON CONFLICT (username) DO UPDATE SET auth_user_id = EXCLUDED.auth_user_id WHERE public.customer_usernames.auth_user_id IS NULL OR public.customer_usernames.auth_user_id = EXCLUDED.auth_user_id; IF NOT EXISTS ( SELECT 1 FROM public.customer_usernames cu WHERE cu.username = NEW.username AND cu.auth_user_id = NEW.user_id ) THEN RAISE EXCEPTION 'Ce pseudo est déjà utilisé par un autre compte. Choisis-en un autre.'; END IF; RETURN NEW; END; $$; DROP TRIGGER IF EXISTS trg_ensure_customer_username_reservation ON public.customer_profiles; CREATE TRIGGER trg_ensure_customer_username_reservation BEFORE INSERT OR UPDATE OF username, user_id ON public.customer_profiles FOR EACH ROW EXECUTE FUNCTION public.ensure_customer_username_reservation(); UPDATE auth.users AS u SET raw_user_meta_data = COALESCE(u.raw_user_meta_data, '{}'::jsonb) || jsonb_build_object('username', 'Mishaa') FROM public.app_settings AS s WHERE s.id = 1 AND s.admin_user_id = u.id; NOTIFY pgrst, 'reload schema';
